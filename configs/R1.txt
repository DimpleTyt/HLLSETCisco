! ===== R1 (ISR4331) - stage 1: base, router-on-a-stick, DHCP =====
enable
configure terminal
hostname R1
no ip domain-lookup
service password-encryption
enable secret Cisco123!
banner motd #HLLSet lab - authorized access only#
username admin privilege 15 secret Admin123!
ip domain-name hllset.lab
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
!
ip access-list standard MGMT-VTY
 permit 192.168.99.0 0.0.0.255
 deny any
exit
line console 0
 logging synchronous
 exec-timeout 15 0
exit
line vty 0 4
 login local
 transport input ssh
 access-class MGMT-VTY in
exit
!
! --- trunk to SW-CORE G0/1 ---
interface GigabitEthernet0/0/0
 description TRUNK to SW-CORE G0/1
 no ip address
 no shutdown
exit
interface GigabitEthernet0/0/0.10
 description VLAN10 USERS
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
exit
interface GigabitEthernet0/0/0.20
 description VLAN20 SERVICES (DATA, DISPATCHER)
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
exit
interface GigabitEthernet0/0/0.30
 description VLAN30 SWARM (FPGA1-5)
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
exit
interface GigabitEthernet0/0/0.40
 description VLAN40 GRID
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
exit
interface GigabitEthernet0/0/0.99
 description VLAN99 MGMT
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
exit
!
! --- DHCP for users ---
ip dhcp excluded-address 192.168.10.1 192.168.10.99
ip dhcp pool USERS
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 domain-name hllset.lab
exit
end
write memory
